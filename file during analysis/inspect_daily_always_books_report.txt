--- book_calendar_mappings ---
CREATE TABLE book_calendar_mappings (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            book_id TEXT NOT NULL,
            calendar_day_id INTEGER,
            ethiopian_month INTEGER,
            ethiopian_day INTEGER,
            is_annual INTEGER NOT NULL DEFAULT 0,
            is_primary_feast INTEGER NOT NULL DEFAULT 1,
            reading_cycle TEXT NOT NULL DEFAULT 'CALENDAR',
            weekday_number INTEGER,
            season_code TEXT,
            app_section TEXT NOT NULL DEFAULT 'TODAYS_FEAST',
            notes TEXT,
            FOREIGN KEY (book_id) REFERENCES books(id) ON DELETE CASCADE,
            FOREIGN KEY (calendar_day_id) REFERENCES calendar_days(id) ON DELETE SET NULL
        )

--- book_seasonal_mappings ---
CREATE TABLE book_seasonal_mappings (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  book_id TEXT,
  season_code TEXT,
  reading_order INTEGER,
  notes TEXT,
  FOREIGN KEY(book_id) REFERENCES books(id),
  FOREIGN KEY(season_code) REFERENCES seasonal_periods(code)
)

--- books ---
CREATE TABLE "books" (
    id TEXT NOT NULL PRIMARY KEY,
    language_id TEXT NOT NULL,
    category_id TEXT NOT NULL,
    title TEXT NOT NULL,
    title_short TEXT,
    description TEXT,
    total_chapters INTEGER NOT NULL DEFAULT 0,
    sort_order INTEGER NOT NULL DEFAULT 0,
    subject_category TEXT DEFAULT '' NOT NULL,
    subject_tag TEXT DEFAULT '' NOT NULL,
    cover_image TEXT DEFAULT '' NOT NULL, main_category_id TEXT REFERENCES main_categories(id), sub_category_id TEXT REFERENCES sub_categories(id),
    FOREIGN KEY (language_id) REFERENCES languages(id),
    FOREIGN KEY (category_id) REFERENCES categories(id)
)

--- calendar_days ---
CREATE TABLE calendar_days (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    ethiopian_month INTEGER NOT NULL,
    day_of_month INTEGER NOT NULL,
    UNIQUE(ethiopian_month, day_of_month),
    FOREIGN KEY (ethiopian_month) REFERENCES ethiopian_months(id)
)

--- chapters ---
CREATE TABLE "chapters" (
    id TEXT NOT NULL PRIMARY KEY,
    book_id TEXT NOT NULL,
    chapter_number INTEGER NOT NULL,
    title TEXT,
